                                                                                                                                                                                                                <script>
                                                                                                                                                                                                                        // Funkcija, kas saņem mērvienības dalītāju (1024 vai 1000)
                                                                                                                                                                                                                                function aprekinat(dalaAr) {
                                                                                                                                                                                                                                            // Iegūstam vērtību no ievades lauka, izmantojot tā ID "sekundes"
                                                                                                                                                                                                                                                        let s = document.getElementById('sekundes').value;
                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                                if (s > 0) {
                                                                                                                                                                                                                                                                                                // APRĒĶINS (Paskaidrojums):
                                                                                                                                                                                                                                                                                                                // 25 Mbps ir 25 000 000 biti sekundē.
                                                                                                                                                                                                                                                                                                                                // 1) 25 * 1 000 000 = biti kopā
                                                                                                                                                                                                                                                                                                                                                // 2) Dalām ar 8 = baiti kopā
                                                                                                                                                                                                                                                                                                                                                                // 3) Dalām ar 'dalaAr' (1024 vai 1000) = gala rezultāts
                                                                                                                                                                                                                                                                                                                                                                                let biti = 25 * 1000000 * s;
                                                                                                                                                                                                                                                                                                                                                                                                let baiti = biti / 8;
                                                                                                                                                                                                                                                                                                                                                                                                                let rezultats = baiti / dalaAr;

                                                                                                                                                                                                                                                                                                                                                                                                                                // Ieliekam rezultātu atbildes laukā, izmantojot tā ID "atbilde"
                                                                                                                                                                                                                                                                                                                                                                                                                                                document.getElementById('atbilde').value = rezultats;
                                                                                                                                                                                                                                                                                                                                                                                                                                                            } else {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                            alert("Lūdzu, ievadiet sekundes!");
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            // Atķeksējam radio pogu atpakaļ, ja nav ievadīts laiks
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            document.getElementsByName('mervieniba').forEach(x => x.checked = false);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    </script>
